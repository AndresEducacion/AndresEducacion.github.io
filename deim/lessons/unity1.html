        // VECTOR2 → posiciones o movimientos en 2D
        Vector2 movimiento2D = new Vector2(1f, 0f);
        transform.Translate(movimiento2D * Time.deltaTime);


        // VECTOR3 → posiciones o movimientos en 3D
        Vector3 movimiento3D = new Vector3(0f, 0f, 1f);
        transform.position += movimiento3D * Time.deltaTime;


        // GAMEOBJECT → referencia a un objeto del juego
        GameObject enemigo;
        enemigo = GameObject.Find("Enemy");


        // TRANSFORM → posición, rotación y escala
        Transform t = transform;
        t.position = new Vector3(0f, 2f, 0f);
        t.localScale = new Vector3(2f, 2f, 2f);


        // QUATERNION → rotaciones
        Quaternion rotacion = Quaternion.Euler(0f, 90f, 0f);
        transform.rotation = rotacion;


        // COLOR → colores
        Renderer r = GetComponent();
        r.material.color = Color.red;


        // TIME → control del tiempo
        float velocidad = 5f;
        transform.Translate(Vector3.forward * velocidad * Time.deltaTime);


        // INPUT.GETAXIS() → movimiento continuo (teclado o mando)
        float horizontal = Input.GetAxis("Horizontal");
        float vertical = Input.GetAxis("Vertical");

        Vector3 movimiento = new Vector3(horizontal, 0f, vertical);
        transform.Translate(movimiento * Time.deltaTime * 5f);


        // INPUT.GETKEY() / GETKEYDOWN() → detección de teclas
        if (Input.GetKey(KeyCode.Space))
        {
            Debug.Log("La tecla espacio está pulsada");
        }

        if (Input.GetKeyDown(KeyCode.Space))
        {
            Debug.Log("Se ha pulsado espacio una vez");
        }

Variables públicas

Ejemplos


        using UnityEngine;

        public class Player : MonoBehaviour
        {
            // VARIABLE PÚBLICA BÁSICA
            public int vidas = 3;

            // VARIABLE PÚBLICA DECIMAL
            public float velocidad = 5f;

            // VARIABLE PÚBLICA BOOLEAN
            public bool estaVivo = true;

            // VARIABLE PÚBLICA DE UNITY
            public GameObject enemigo;

            // VARIABLE PÚBLICA VECTOR
            public Vector3 posicionInicial;

            void Start()
            {
                // Accedemos a las variables públicas
                Debug.Log("Vidas iniciales: " + vidas);

                // Usamos la velocidad para mover el objeto
                transform.Translate(Vector3.forward * velocidad * Time.deltaTime);

                // Colocamos el objeto en la posición indicada
                transform.position = posicionInicial;

                // Accedemos al GameObject asignado desde el Inspector
                if (enemigo != null)
                {
                    Debug.Log("Enemigo asignado: " + enemigo.name);
                }
            }
        }

Primer script en Unity


        using UnityEngine;

        public class EjemploMono : MonoBehaviour
        {
            //TIPOS DE VARIABLES
            int numeroEntero = 10;
            float numeroDecimal = 3.5f;
            bool activo = true;
            string nombre = "Unity";
            char letra = 'U';

            int[] listaNumeros = { 2, 4, 6, 8 };
            string[] listaSaludos = {"heyy","hola","klk"};


            void Start()
            {
                Debug.Log("Start ejecutado");

                // Operadores
                int resultado = numeroEntero * 2;
                float division = numeroDecimal / 2f; // noten que al dividir numeros decimales necesitamos que divisor y dividendo sean decimales (con la "f")


                // Imprime en consola los resultados
                Debug.Log("Resultado multiplicación: " + resultado);
                Debug.Log("Resultado división: " + division);

                // If con operador lógico
                if (activo && numeroEntero > 5) // al poner una bool en una condicional, se asume que tiene que ser "true" para que se suceda la acción. En este caso lo es (pusimos true al declararla)
                {
                    Debug.Log("Condición del IF cumplida.");
                }

                // Bucle for
                for (int i = 0; i < listaNumeros.Length; i++)
                {
                    Debug.Log("Número en lista: " + listaNumeros[i]);
                }

                // Llamada a una función
                MostrarMensaje("¡Hola desde la función!"); // Las funciones se pueden llamar en cualquier momento del código, da igual si las definimos antes o después.
            }

            // Update: se ejecuta una vez por frame
            void Update()
            {
                // Ejemplo simple dentro de Update:
                if (Input.GetKeyDown(KeyCode.Space))
                {
                    Debug.Log("Has presionado la barra espaciadora.");
                }
            }

            // Función personalizada
            void MostrarMensaje(string texto)
            {
                Debug.Log("Mensaje: " + texto);
            }
        }
